create table public.prompt_visual_reference_sets (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  business_key text not null check (char_length(business_key) between 1 and 700),
  guidance text not null default '' check (char_length(guidance) <= 1200),
  analysis jsonb,
  analysis_signature text,
  analysis_model text,
  analyzed_at timestamptz,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (user_id, business_key)
);

create table public.prompt_visual_references (
  id uuid primary key default gen_random_uuid(),
  set_id uuid not null references public.prompt_visual_reference_sets(id) on delete cascade,
  user_id uuid not null references auth.users(id) on delete cascade,
  file_hash text not null check (file_hash ~ '^[0-9a-f]{64}$'),
  storage_path text not null,
  original_name text not null check (char_length(original_name) between 1 and 240),
  mime_type text not null default 'image/webp' check (mime_type = 'image/webp'),
  byte_size integer not null check (byte_size > 0 and byte_size <= 3145728),
  width integer not null check (width > 0 and width <= 1600),
  height integer not null check (height > 0 and height <= 12000),
  note text not null default '' check (char_length(note) <= 500),
  sort_order smallint not null check (sort_order between 0 and 4),
  created_at timestamptz not null default now(),
  unique (set_id, file_hash),
  unique (set_id, sort_order),
  unique (storage_path)
);

create index prompt_visual_reference_sets_user_business_idx
  on public.prompt_visual_reference_sets(user_id, business_key);

create index prompt_visual_references_set_order_idx
  on public.prompt_visual_references(set_id, sort_order);
