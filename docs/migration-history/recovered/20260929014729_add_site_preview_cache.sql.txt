create table if not exists public.site_previews (
  id uuid primary key default extensions.gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  business_key text not null,
  company text not null,
  prompt_version text not null,
  preview_version uuid not null default extensions.gen_random_uuid(),
  preview_spec jsonb not null,
  generated_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (user_id, business_key)
);

create index if not exists site_previews_user_business_idx
  on public.site_previews(user_id, business_key);

alter table public.site_previews enable row level security;

create policy site_previews_select_own
on public.site_previews
for select
to authenticated
using (
  (select auth.uid()) = user_id
  and exists (
    select 1
    from public.private_access pa
    where pa.user_id = (select auth.uid())
  )
);

create policy site_previews_insert_own
on public.site_previews
for insert
to authenticated
with check (
  (select auth.uid()) = user_id
  and exists (
    select 1
    from public.private_access pa
    where pa.user_id = (select auth.uid())
  )
);

create policy site_previews_update_own
on public.site_previews
for update
to authenticated
using (
  (select auth.uid()) = user_id
  and exists (
    select 1
    from public.private_access pa
    where pa.user_id = (select auth.uid())
  )
)
with check (
  (select auth.uid()) = user_id
  and exists (
    select 1
    from public.private_access pa
    where pa.user_id = (select auth.uid())
  )
);

create policy site_previews_delete_own
on public.site_previews
for delete
to authenticated
using (
  (select auth.uid()) = user_id
  and exists (
    select 1
    from public.private_access pa
    where pa.user_id = (select auth.uid())
  )
);

revoke all on table public.site_previews from anon;
grant select, insert, update, delete on table public.site_previews to authenticated;
