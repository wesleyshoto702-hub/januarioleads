-- ROLLBACK CANDIDATE — EMERGENCY ONLY; NOT A DEPLOYABLE MIGRATION
--
-- Restores the *original*, owner-only RLS expressions observed in
-- pg_policies on 2026-11-05. THIS WOULD REOPEN THE CONFIRMED REVOKED-JWT
-- ACCESS GAP. Use only for a service-restoration incident after review.
-- Never execute directly on production without an approved incident plan.
-- No GRANTs, table rows, functions, owners, or auth records are modified.
-- Like the forward candidate, all changes are transactional.
BEGIN;
SET LOCAL lock_timeout = '5s';
SET LOCAL statement_timeout = '30s';

DROP POLICY IF EXISTS "Users can read own lead payments"
  ON public.lead_payments;
CREATE POLICY "Users can read own lead payments"
  ON public.lead_payments FOR SELECT TO authenticated
  USING ((SELECT auth.uid()) = user_id);

DROP POLICY IF EXISTS "Users can read own open now cache"
  ON public.open_now_niche_cache;
CREATE POLICY "Users can read own open now cache"
  ON public.open_now_niche_cache FOR SELECT TO authenticated
  USING ((SELECT auth.uid()) = user_id);

DROP POLICY IF EXISTS "Users can read own places search cache"
  ON public.places_search_cache;
CREATE POLICY "Users can read own places search cache"
  ON public.places_search_cache FOR SELECT TO authenticated
  USING ((SELECT auth.uid()) = user_id);

DROP POLICY IF EXISTS "Users can insert own places search cache"
  ON public.places_search_cache;
CREATE POLICY "Users can insert own places search cache"
  ON public.places_search_cache FOR INSERT TO authenticated
  WITH CHECK ((SELECT auth.uid()) = user_id);

DROP POLICY IF EXISTS "Users can update own places search cache"
  ON public.places_search_cache;
CREATE POLICY "Users can update own places search cache"
  ON public.places_search_cache FOR UPDATE TO authenticated
  USING ((SELECT auth.uid()) = user_id)
  WITH CHECK ((SELECT auth.uid()) = user_id);

COMMIT;
